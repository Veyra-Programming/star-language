Developer Quickstart

        
Get Started with Star.js in 60 Seconds

        

          Zero build step needed. Just drop in the standalone script tag, define your reactive components, and enjoy
          full disruption-tolerant digital continuity.
        

      


      

        

          index.html
            (React Hooks + Star.html CDN Example)
          
        

        
<!DOCTYPE html>
<html>
<head>
  <title>My Star App</title>
  <link rel="stylesheet" href="style.css">
  <script src="star.js"></script>
</head>
<body>
  <div id="app"></div>

  <script>
    const { useState, useEffect, useODCS, html, createApp } = Star;

    function MyApp() {
      const [count, setCount] = useState(0);
      const odcs = useODCS();

      const handleClick = async () => {
        setCount(c => c + 1);
        // Mint Layer 4 ZIL receipt in <16ms!
        await odcs.preserve("CLICK", { count: count + 1 });
      };

      return html`
        <div class="star-card">
          <h2>🌟 Star.js Disruption-Tolerant App</h2>
          <p>Mode: <strong>${odcs.mode}</strong> | Clicks: <strong>${count}</strong></p>
          <button class="star-btn star-btn-primary" onClick=${handleClick}>
            Preserve Intent
          </button>
        </div>
      `;
    }

    createApp(MyApp).mount("#app");
  </script>
</body>
</html>
